\maketitle

A citation appears in the text with the \texttt{\\cite} command. For example,
Lamport's book describes the LaTeX system and its document preparation model
\cite{lamport1994}.

\begin{thebibliography}{9}
  \bibitem{lamport1994}
  Leslie Lamport.
  \emph{LaTeX: A Document Preparation System}.
  2nd edition, Addison-Wesley, 1994.
\end{thebibliography}

\end{document}
